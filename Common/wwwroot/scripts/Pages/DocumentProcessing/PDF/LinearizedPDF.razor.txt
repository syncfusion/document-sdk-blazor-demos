@code {
    private readonly Dictionary<string, MemoryStream> fileData = new Dictionary<string, MemoryStream>();
    private string selectedFileName = "pdf-succinctly.pdf";

    private string errorMessage = string.Empty;

    private async Task OnInputFileChange(InputFileChangeEventArgs args)
    {
        errorMessage = string.Empty;
        ClearFileData();
        IBrowserFile file = args.File;
        const long MAX_FILE_SIZE = 10 * 1024 * 1024;
        if (file.Size > MAX_FILE_SIZE)
        {
            errorMessage = string.Format("The selected PDF file exceeds the maximum allowed size of 10 MB. File size: {0:F2} MB.", file.Size / (1024.0 * 1024.0));
            selectedFileName = "pdf-succinctly.pdf";
            return;
        }
        MemoryStream stream = new MemoryStream();
        await file.OpenReadStream(maxAllowedSize: MAX_FILE_SIZE).CopyToAsync(stream);
        stream.Position = 0;
        fileData.Add(file.Name, stream);
        selectedFileName = file.Name;
    }

    protected async Task LinearizePdf(MouseEventArgs args)
    {
        await LoadDefaultFile();

        LinearizedPDFService service = new LinearizedPDFService(fileData);
        MemoryStream documentStream = service.LinearizePdf();
        service.Close();
        await JS.SaveAs(selectedFileName, documentStream.ToArray());
        documentStream.Dispose();
    }

    private async Task LoadDefaultFile()
    {
        if (fileData.Count > 0)
        {
            return;
        }

        string fileDataPath = NavigationManager.BaseUri + SampleService?.WebAssetsPath;
        byte[] data = await Http.GetByteArrayAsync(fileDataPath + "data/pdf/pdf-succinctly.pdf");
        fileData.Add("pdf-succinctly.pdf", new MemoryStream(data));
    }

    private void ClearFileData()
    {
        foreach (MemoryStream stream in fileData.Values)
        {
            stream.Dispose();
        }
        fileData.Clear();
    }
}
