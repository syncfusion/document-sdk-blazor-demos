@code {
    MemoryStream? documentStream;
    //Hidden:Lines
    
    Dictionary<string, MemoryStream> fileData = new Dictionary<string, MemoryStream>();
    
    //End:Hidden

    /// <summary>
    /// Convert Markdown to PDF
    /// </summary>
    public async Task GenerateDocument(MouseEventArgs args)
    {
        await LoadTemplate();
        MarkdownToPDFService service = new MarkdownToPDFService(fileData);
        documentStream = service.MarkdownToPDF("Generate Document");
        service.Close();
		await JS.SaveAs("MarkdownToWord.pdf", documentStream.ToArray());
    }
    /// <summary>
    /// Create and download the template Markdown document
    /// </summary>
    public async Task ViewTemplate(MouseEventArgs args)
    {
        await LoadTemplate();
        MarkdownToPDFService service = new MarkdownToPDFService(fileData);
        documentStream = service.MarkdownToPDF("View Template");
        service.Close();
        await JS.SaveAs("MarkdownToWord.md", documentStream.ToArray());        
    }
    /// <summary>
    /// Load the template document
    /// </summary>
    public async Task LoadTemplate()
    {
        if (fileData != null && fileData.Count == 0 && SampleService?.WebAssetsPath != null)
        {
            string fileDataPath = NavigationManager.BaseUri + SampleService.WebAssetsPath;
            byte[] data = await Http.GetByteArrayAsync(fileDataPath + "data/markdown/markdown-to-word.md");
            fileData.Add("markdown-to-word.md", new MemoryStream(data));
        }
    }
}
